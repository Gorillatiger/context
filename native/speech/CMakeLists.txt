cmake_minimum_required(VERSION 3.22)
project(context_speech LANGUAGES C CXX)

if(NOT DEFINED WHISPER_SOURCE_DIR OR NOT EXISTS "${WHISPER_SOURCE_DIR}/include/whisper.h")
  message(FATAL_ERROR "Use node desktop/build-speech.mjs to obtain the verified whisper.cpp source.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Portable CPU binary" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "No extra OpenMP runtime dependency" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "Include Metal shaders in the executable" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "No dynamic backend lookup" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper)
find_package(Threads REQUIRED)
add_executable(speech-helper helper.cpp)
target_include_directories(speech-helper PRIVATE "${WHISPER_SOURCE_DIR}/examples")
target_link_libraries(speech-helper PRIVATE whisper Threads::Threads)
if(WIN32)
  target_link_libraries(speech-helper PRIVATE psapi)
  target_compile_definitions(speech-helper PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(MSVC)
  set_property(TARGET speech-helper PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(speech-helper PRIVATE /W4)
else()
  target_compile_options(speech-helper PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(speech-helper PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
